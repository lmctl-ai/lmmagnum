cmake_minimum_required(VERSION 3.24)
project(lmmagnum VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(LMMAGNUM_FULL_STATIC "Link the complete executable statically" ON)

set(LMMAGNUM_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmmagnum.dsl.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LMMAGNUM_SCHEMA}")
file(READ "${LMMAGNUM_SCHEMA}" LMMAGNUM_SCHEMA_HEX HEX)
string(LENGTH "${LMMAGNUM_SCHEMA_HEX}" LMMAGNUM_SCHEMA_HEX_LENGTH)
math(EXPR LMMAGNUM_SCHEMA_LAST_BYTE "${LMMAGNUM_SCHEMA_HEX_LENGTH} - 2")
set(LMMAGNUM_SCHEMA_INITIALIZER "")
foreach(offset RANGE 0 ${LMMAGNUM_SCHEMA_LAST_BYTE} 2)
  string(SUBSTRING "${LMMAGNUM_SCHEMA_HEX}" ${offset} 2 byte)
  string(APPEND LMMAGNUM_SCHEMA_INITIALIZER "0x${byte},")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/schema_embedded.hpp"
  "#pragma once\n#include <cstddef>\nnamespace lmmagnum { inline constexpr unsigned char embedded_schema[] = {${LMMAGNUM_SCHEMA_INITIALIZER}}; inline constexpr std::size_t embedded_schema_size = sizeof(embedded_schema); }\n")

# Magnum + Corrade are built from the pinned clones in third_party/ as static
# libraries. No GL, windowing, audio or plugin code: the host rasterizes in software.
set(CORRADE_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(MAGNUM_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_STATIC_PIC OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CORRADE_BUILD_DEPRECATED OFF CACHE BOOL "" FORCE)
set(CORRADE_WITH_INTERCONNECT OFF CACHE BOOL "" FORCE)
set(CORRADE_WITH_PLUGINMANAGER ON CACHE BOOL "" FORCE)
set(CORRADE_WITH_TESTSUITE OFF CACHE BOOL "" FORCE)
set(CORRADE_WITH_MAIN OFF CACHE BOOL "" FORCE)
set(CORRADE_WITH_RC ON CACHE BOOL "" FORCE)
add_subdirectory(third_party/corrade EXCLUDE_FROM_ALL)
set(MAGNUM_BUILD_DEPRECATED OFF CACHE BOOL "" FORCE)
set(MAGNUM_WITH_GL OFF CACHE BOOL "" FORCE)
set(MAGNUM_WITH_AUDIO OFF CACHE BOOL "" FORCE)
set(MAGNUM_WITH_DEBUGTOOLS OFF CACHE BOOL "" FORCE)
set(MAGNUM_WITH_SCENEGRAPH OFF CACHE BOOL "" FORCE)
set(MAGNUM_WITH_SHADERS OFF CACHE BOOL "" FORCE)
set(MAGNUM_WITH_TEXT OFF CACHE BOOL "" FORCE)
set(MAGNUM_WITH_TEXTURETOOLS OFF CACHE BOOL "" FORCE)
set(MAGNUM_WITH_VK OFF CACHE BOOL "" FORCE)
set(MAGNUM_WITH_PRIMITIVES ON CACHE BOOL "" FORCE)
set(MAGNUM_WITH_MESHTOOLS ON CACHE BOOL "" FORCE)
set(MAGNUM_WITH_TRADE ON CACHE BOOL "" FORCE)
add_subdirectory(third_party/magnum EXCLUDE_FROM_ALL)

execute_process(COMMAND git -C "${CMAKE_CURRENT_SOURCE_DIR}/third_party/magnum" rev-parse --short=12 HEAD OUTPUT_VARIABLE LMMAGNUM_MAGNUM_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT LMMAGNUM_MAGNUM_COMMIT)
  set(LMMAGNUM_MAGNUM_COMMIT "unknown")
endif()

add_executable(lmmagnum cpp/main.cpp cpp/scene.cpp cpp/render.cpp cpp/atomic-file.cpp)
set_target_properties(lmmagnum PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}")
target_compile_definitions(lmmagnum PRIVATE LMMAGNUM_VERSION="${PROJECT_VERSION}" LMMAGNUM_MAGNUM_COMMIT="${LMMAGNUM_MAGNUM_COMMIT}")
target_include_directories(lmmagnum PRIVATE cpp "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(lmmagnum PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(lmmagnum PRIVATE Magnum::Magnum Magnum::Trade Magnum::Primitives Magnum::MeshTools)
if(NOT WIN32 AND NOT APPLE)
  target_link_libraries(lmmagnum PRIVATE pthread dl m rt)
endif()
# "Clang" also matches AppleClang; macOS has no -static/-static-libstdc++.
if((CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang") AND NOT APPLE)
  target_link_options(lmmagnum PRIVATE -static-libstdc++ -static-libgcc)
  if(LMMAGNUM_FULL_STATIC)
    target_link_options(lmmagnum PRIVATE -static)
  endif()
endif()
